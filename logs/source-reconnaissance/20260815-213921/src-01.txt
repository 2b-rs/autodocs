   238	def _pdf_string_bytes(raw: bytes) -> bytes:
   239	    """Decode PDF literal/hex string syntax while preserving character codes."""
   240	    if raw.startswith(b"<"):
   241	        digits = re.sub(rb"[^0-9A-Fa-f]", b"", raw[1:-1])
   242	        if len(digits) % 2:
   243	            digits += b"0"
   244	        return bytes.fromhex(digits.decode("ascii"))
   245	    body, out, i = raw[1:-1], bytearray(), 0
   246	    simple = {ord("n"): 10, ord("r"): 13, ord("t"): 9, ord("b"): 8,
   247	              ord("f"): 12, ord("("): 40, ord(")"): 41, ord("\\"): 92}
   248	    while i < len(body):
   249	        if body[i] != 92:
   250	            out.append(body[i]); i += 1; continue
   251	        i += 1
   252	        if i >= len(body):
   253	            break
   254	        ch = body[i]
   255	        if ch in simple:
   256	            out.append(simple[ch]); i += 1
   257	        elif ch in (10, 13):
   258	            i += 1
   259	            if ch == 13 and i < len(body) and body[i] == 10:
   260	                i += 1
   261	        elif 48 <= ch <= 55:
   262	            j = i
   263	            while j < len(body) and j < i + 3 and 48 <= body[j] <= 55:
   264	                j += 1
   265	            out.append(int(body[i:j], 8) & 255); i = j
   266	        else:
   267	            out.append(ch); i += 1
   268	    return bytes(out)

   278	def _parse_tounicode(data: bytes) -> dict:
   279	    """Parse bfchar and bfrange entries from a PDF ToUnicode CMap."""
   280	    result = {}
   281	    for block in re.findall(rb"beginbfchar(.*?)endbfchar", data, re.S):
   282	        for src, dst in re.findall(rb"<([0-9A-Fa-f]+)>\s*<([0-9A-Fa-f]+)>", block):
   283	            result[bytes.fromhex(src.decode())] = _unicode_value(bytes.fromhex(dst.decode()))
   284	    for block in re.findall(rb"beginbfrange(.*?)endbfrange", data, re.S):
   285	        entries = re.findall(rb"<([0-9A-Fa-f]+)>\s*<([0-9A-Fa-f]+)>\s*(<([0-9A-Fa-f]+)>|\[(.*?)\])", block, re.S)
   286	        for lo, hi, target, sequential, array in entries:
   287	            lo_b, hi_b = bytes.fromhex(lo.decode()), bytes.fromhex(hi.decode())
   288	            width = len(lo_b); first, last = int.from_bytes(lo_b, "big"), int.from_bytes(hi_b, "big")
   289	            if sequential:
   290	                dst = bytes.fromhex(sequential.decode()); base = int.from_bytes(dst, "big"); dwidth = len(dst)
   291	                values = [(base + offset).to_bytes(dwidth, "big") for offset in range(last-first+1)]
   292	            else:
   293	                values = [bytes.fromhex(x.decode()) for x in re.findall(rb"<([0-9A-Fa-f]+)>", array)]
   294	            for offset, value in enumerate(values[:last-first+1]):
   295	                result[(first + offset).to_bytes(width, "big")] = _unicode_value(value)
   296	    return result

   299	def _decode_pdf_string(raw: bytes, cmap=None) -> str:
   300	    data = _pdf_string_bytes(raw)
   301	    if not cmap:
   302	        if data[:2] in (b"\xfe\xff", b"\xff\xfe"):
   303	            return data.decode("utf-16", "replace")
   304	        return data.decode("latin-1", "replace")
   305	    lengths = sorted({len(key) for key in cmap}, reverse=True)
   306	    out, pos = [], 0
   307	    while pos < len(data):
   308	        for width in lengths:
   309	            token = data[pos:pos + width]
   310	            if len(token) == width and token in cmap:
   311	                out.append(cmap[token]); pos += width; break
   312	        else:
   313	            out.append(bytes([data[pos]]).decode("latin-1", "replace")); pos += 1
   314	    return "".join(out)

   327	def _content_to_text(content: bytes, fonts=None) -> str:
   328	    """Content stream to text, applying the active font's ToUnicode CMap."""
   329	    parts, active = [], None
   330	    fonts = fonts or {}
   331	    for m in _OPS.finditer(content):
   332	        if m.group("font"):
   333	            active = m.group("font")[1:].decode("ascii", "replace")
   334	        elif m.group("brk"):
   335	            parts.append("\n")
   336	        elif m.group("lit") or m.group("hex"):
   337	            parts.append(_decode_pdf_string(m.group("lit") or m.group("hex"), fonts.get(active)))
   338	        else:
   339	            for item in _ARR_ITEM.finditer(m.group("arr")):
   340	                tok = item.group(0)
   341	                if tok[:1] in (b"(", b"<"):
   342	                    parts.append(_decode_pdf_string(tok, fonts.get(active)))
   343	                else:
   344	                    try:
   345	                        if float(tok) <= -100:
   346	                            parts.append(" ")
   347	                    except ValueError:
   348	                        pass
   349	    return "".join(parts)

   352	def _inflate(raw: bytes, body: bytes) -> bytes:
   353	    """FlateDecode inkl. PNG-Predictor (/Predictor >= 10), sonst Rohdaten."""
   354	    if b"/FlateDecode" not in body:
   355	        return raw
   356	    try:
   357	        data = zlib.decompress(raw)
   358	    except zlib.error:
   359	        try:
   360	            data = zlib.decompressobj().decompress(raw)
   361	        except zlib.error:
   362	            return b""
   363	    m = re.search(rb"/Predictor\s+(\d+)", body)
   364	    if not m or int(m.group(1)) < 10:
   365	        return data
   366	    cols = int((re.search(rb"/Columns\s+(\d+)", body) or [b"", b"1"])[1])
   367	    colors = int((re.search(rb"/Colors\s+(\d+)", body) or [b"", b"1"])[1])
   368	    bpc = int((re.search(rb"/BitsPerComponent\s+(\d+)", body) or [b"", b"8"])[1])
   369	    bpp = max(1, colors * bpc // 8)
   370	    rowlen = cols * colors * bpc // 8
   371	    out, prev = bytearray(), bytearray(rowlen)
   372	    pos = 0
   373	    while pos + 1 + rowlen <= len(data):
   374	        ft = data[pos]
   375	        row = bytearray(data[pos + 1 : pos + 1 + rowlen])
   376	        pos += 1 + rowlen
   377	        for i in range(rowlen):
   378	            a = row[i - bpp] if i >= bpp else 0
   379	            b = prev[i]
   380	            c = prev[i - bpp] if i >= bpp else 0
   381	            if ft == 1:
   382	                row[i] = (row[i] + a) & 0xFF
   383	            elif ft == 2:
   384	                row[i] = (row[i] + b) & 0xFF
   385	            elif ft == 3:
   386	                row[i] = (row[i] + (a + b) // 2) & 0xFF
   387	            elif ft == 4:
   388	                pp = a + b - c
   389	                pa, pb, pc = abs(pp - a), abs(pp - b), abs(pp - c)
   390	                pr = a if (pa <= pb and pa <= pc) else (b if pb <= pc else c)
   391	                row[i] = (row[i] + pr) & 0xFF
   392	        out += row
   393	        prev = row
   394	    return bytes(out)

   397	def _collect_objects(data: bytes) -> dict:
   398	    """Alle Objekte einsammeln — klassisch und aus Objektstroemen (/ObjStm).
   399	
   400	    Moderne PDFs (auch die AUTOSAR-Dokumente) legen die meisten Objekte in
   401	    komprimierten Objektstroemen ab; ohne deren Aufloesung findet man keine
   402	    einzige Seite.
   403	    """
   404	    bodies, streams = {}, {}
   405	    for m in re.finditer(rb"(\d+)\s+\d+\s+obj(.*?)endobj", data, re.S):
   406	        num, body = int(m.group(1)), m.group(2)
   407	        bodies[num] = body
   408	        sm = re.search(rb"stream\r?\n(.*?)\r?\n?endstream", body, re.S)
   409	        if sm:
   410	            streams[num] = _inflate(sm.group(1), body)
   411	    for num, body in list(bodies.items()):
   412	        if not re.search(rb"/Type\s*/ObjStm", body) or num not in streams:
   413	            continue
   414	        payload = streams[num]
   415	        n = int((re.search(rb"/N\s+(\d+)", body) or [b"", b"0"])[1])
   416	        first = int((re.search(rb"/First\s+(\d+)", body) or [b"", b"0"])[1])
   417	        header = payload[:first].split()
   418	        for i in range(min(n, len(header) // 2)):
   419	            onum, off = int(header[2 * i]), int(header[2 * i + 1])
   420	            ende = (int(header[2 * i + 3]) + first) if 2 * i + 3 < len(header) else len(payload)
   421	            bodies.setdefault(onum, payload[first + off : ende])
   422	    return bodies, streams

   425	def _builtin_pdf_pages(path: Path) -> list:
   426	    """Minimalparser ohne Fremdbibliothek.
   427	
   428	    Seiten werden ueber den Seitenbaum (/Type /Pages -> /Kids) geordnet, damit
   429	    die Reihenfolge der Dokumentreihenfolge entspricht; nur falls kein Baum
   430	    auffindbar ist, greift die Objektnummern-Reihenfolge.
   431	    """
   432	    data = path.read_bytes()
   433	    bodies, streams = _collect_objects(data)
   434	    seiten = [n for n, b in bodies.items() if re.search(rb"/Type\s*/Page[^s]", b + b" ")]
   435	
   436	    reihenfolge, gesehen = [], set()
   437	
   438	    def kids_of(num, tiefe=0):
   439	        if tiefe > 64 or num in gesehen:
   440	            return
   441	        gesehen.add(num)
   442	        body = bodies.get(num, b"")
   443	        if re.search(rb"/Type\s*/Page[^s]", body + b" "):
   444	            reihenfolge.append(num)
   445	            return
   446	        km = re.search(rb"/Kids\s*\[(.*?)\]", body, re.S)
   447	        if km:
   448	            for kid in re.findall(rb"(\d+)\s+\d+\s+R", km.group(1)):
   449	                kids_of(int(kid), tiefe + 1)
   450	
   451	    wurzeln = [n for n, b in bodies.items()
   452	               if re.search(rb"/Type\s*/Pages", b) and b"/Parent" not in b]
   453	    for w in wurzeln:
   454	        kids_of(w)
   455	    if not reihenfolge:
   456	        reihenfolge = sorted(seiten)
   457	
   458	    ergebnis = []
   459	    for num in reihenfolge:
   460	        body = bodies.get(num, b"")
   461	        refs = [int(r) for r in re.findall(rb"/Contents\s+(\d+)\s+\d+\s+R", body)]
   462	        arr = re.search(rb"/Contents\s*\[(.*?)\]", body, re.S)
   463	        if arr:
   464	            refs += [int(r) for r in re.findall(rb"(\d+)\s+\d+\s+R", arr.group(1))]
   465	        resource = body
   466	        rm = re.search(rb"/Resources\s+(\d+)\s+\d+\s+R", body)
   467	        if rm:
   468	            resource = bodies.get(int(rm.group(1)), b"")
   469	        fonts = {}
   470	        fm = re.search(rb"/Font\s*<<(.*?)>>", resource, re.S)
   471	        if fm:
   472	            for alias, font_ref in re.findall(rb"/([A-Za-z0-9_.+-]+)\s+(\d+)\s+\d+\s+R", fm.group(1)):
   473	                font_body = bodies.get(int(font_ref), b"")
   474	                tm = re.search(rb"/ToUnicode\s+(\d+)\s+\d+\s+R", font_body)
   475	                if tm and int(tm.group(1)) in streams:
   476	                    fonts[alias.decode("ascii", "replace")] = _parse_tounicode(streams[int(tm.group(1))])
   477	        ergebnis.append("".join(_content_to_text(streams[r], fonts) for r in refs if r in streams))
   478	    return ergebnis

   484	def _matrix_values(value) -> list[float] | None:
   485	    """Return a stable six-value PDF matrix without retaining backend objects."""
   486	    if not value or len(value) < 6:
   487	        return None
   488	    return [round(float(item), 6) for item in value[:6]]

   491	def _effective_text_position(current_matrix, text_matrix) -> tuple[float, float] | None:
   492	    """Transform a text-matrix origin into page coordinates."""
   493	    cm = _matrix_values(current_matrix)
   494	    tm = _matrix_values(text_matrix)
   495	    if cm is None or tm is None:
   496	        return None
   497	    x = tm[4] * cm[0] + tm[5] * cm[2] + cm[4]
   498	    y = tm[4] * cm[1] + tm[5] * cm[3] + cm[5]
   499	    return round(x, 6), round(y, 6)

   502	def _classify_line_layout(line: dict, spans_by_id: dict[str, dict]) -> dict:
   503	    """Describe aligned horizontal cells without changing reading order."""
   504	    groups = []
   505	    for span_id in line["ordered_span_ids"]:
   506	        span = spans_by_id[span_id]
   507	        x = float(span["position"][0])
   508	        if not groups or abs(x - groups[-1]["x"]) > 1.0:
   509	            groups.append({"x": round(x, 6), "span_ids": [span_id]})
   510	        else:
   511	            groups[-1]["span_ids"].append(span_id)
   512	    gaps = [round(right["x"] - left["x"], 6) for left, right in zip(groups, groups[1:])]
   513	    font_size = max((float(spans_by_id[s].get("font_size") or 0) for s in line["span_ids"]), default=1.0)
   514	    threshold = max(36.0, font_size * 3.0)
   515	    boundaries = [index + 1 for index, gap in enumerate(gaps) if gap >= threshold]
   516	    cells = []
   517	    start = 0
   518	    for end in [*boundaries, len(groups)]:
   519	        ids = [sid for group in groups[start:end] for sid in group["span_ids"]]
   520	        if ids:
   521	            cells.append({"id": f"{line['id']}-c{len(cells) + 1}", "span_ids": ids,
   522	                          "x_range": [groups[start]["x"], groups[end - 1]["x"]]})
   523	        start = end
   524	    return {
   525	        "kind": "cell-candidate" if len(cells) > 1 else "single-flow",
   526	        "cells": cells,
   527	        "gaps": gaps,
   528	        "cell_gap_threshold": round(threshold, 6),
   529	    }

   532	def _promote_repeated_cell_patterns(lines: list[dict]) -> None:
   533	    """Promote nearby repeated alignments into explicit, stable table regions."""
   534	    candidates = []
   535	    for line_index, line in enumerate(lines):
   536	        layout = line["layout"]
   537	        layout["table_region_id"] = None
   538	        if layout["kind"] != "cell-candidate":
   539	            layout["alignment_support"] = 0
   540	            continue
   541	        pattern = tuple(round(cell["x_range"][0], 1) for cell in layout["cells"])
   542	        layout["alignment_pattern"] = list(pattern)
   543	        candidates.append((line_index, line, pattern))
   544	    for line_index, line, pattern in candidates:
   545	        supporting = []
   546	        for other_line_index, other, other_pattern in candidates:
   547	            if len(other_pattern) != len(pattern):
   548	                continue
   549	            if max(abs(a - b) for a, b in zip(pattern, other_pattern)) > 1.0:
   550	                continue
   551	            if abs(line_index - other_line_index) > 8:
   552	                continue
   553	            supporting.append(other["id"])
   554	        layout = line["layout"]
   555	        layout["alignment_support"] = len(supporting)
   556	        layout["supporting_line_ids"] = supporting
   557	        layout["kind"] = (
   558	            "table-row-candidate" if len(supporting) >= 2
   559	            else "isolated-gap-candidate"
   560	        )
   561	
   562	    regions = []
   563	    for line_index, line, pattern in candidates:
   564	        if line["layout"]["kind"] != "table-row-candidate":
   565	            continue
   566	        matching = None
   567	        for region in reversed(regions):
   568	            same_shape = len(region["pattern"]) == len(pattern)
   569	            aligned = same_shape and max(abs(a - b) for a, b in zip(region["pattern"], pattern)) <= 1.0
   570	            if aligned and line_index - region["last_line_index"] <= 8:
   571	                matching = region
   572	                break
   573	        if matching is None:
   574	            matching = {"id": f"table-r{len(regions) + 1}", "pattern": pattern,
   575	                        "last_line_index": line_index, "lines": []}
   576	            regions.append(matching)
   577	        matching["last_line_index"] = line_index
   578	        matching["lines"].append(line)
   579	    for region in regions:
   580	        if len(region["lines"]) < 2:
   581	            continue
   582	        for line in region["lines"]:
   583	            layout = line["layout"]
   584	            layout["kind"] = "table-row"
   585	            layout["table_region_id"] = region["id"]
   586	            for cell_index, cell in enumerate(layout["cells"], 1):
   587	                cell["column_index"] = cell_index - 1
   588	                cell["region_cell_id"] = f"{region['id']}-c{cell_index}"

   591	def _classify_repeated_margin_bands(pages: list[dict]) -> None:
   592	    """Mark coordinate-stable margin lines while preserving their evidence."""
   593	    if not pages:
   594	        return
   595	    page_count = len(pages)
   596	    minimum_support = max(2, int(page_count * 0.8 + 0.999999))
   597	    bands = {}
   598	    for page in pages:
   599	        seen = set()
   600	        for line in page["lines"]:
   601	            y = round(float(line["baseline_y"]), 0)
   602	            if y <= 0:
   603	                continue
   604	            seen.add(y)
   605	        for y in seen:
   606	            bands[y] = bands.get(y, 0) + 1
   607	    repeated = {y: support for y, support in bands.items() if support >= minimum_support}
   608	    for page in pages:
   609	        baselines = [float(line["baseline_y"]) for line in page["lines"] if line["baseline_y"] > 0]
   610	        top = max(baselines, default=0.0)
   611	        for line in page["lines"]:
   612	            y = round(float(line["baseline_y"]), 0)
   613	            support = repeated.get(y, 0)
   614	            line["margin_band"] = None
   615	            line["margin_band_support"] = support
   616	            if not support:
   617	                continue
   618	            if line["baseline_y"] <= 72:
   619	                line["margin_band"] = "footer"
   620	            elif top and line["baseline_y"] >= top - 36:
   621	                line["margin_band"] = "header"
   622	    _mark_varying_margin_spans(pages)

   625	def _mark_varying_margin_spans(pages: list[dict]) -> None:
   626	    """Separate page-varying margin text (page numbers) from boilerplate."""
   627	    observed = {}
   628	    for page in pages:
   629	        spans = {span["id"]: span for span in page.get("spans", [])}
   630	        for line in page["lines"]:
   631	            if not line["margin_band"]:
   632	                continue
   633	            for index, span_id in enumerate(line.get("ordered_span_ids", [])):
   634	                key = (line["margin_band"], round(float(line["baseline_y"]), 0), index)
   635	                text = spans.get(span_id, {}).get("text", "").strip()
   636	                observed.setdefault(key, set()).add(text)
   637	    for page in pages:
   638	        spans = {span["id"]: span for span in page.get("spans", [])}
   639	        for line in page["lines"]:
   640	            roles = {}
   641	            if line["margin_band"]:
   642	                for index, span_id in enumerate(line.get("ordered_span_ids", [])):
   643	                    key = (line["margin_band"], round(float(line["baseline_y"]), 0), index)
   644	                    variants = len(observed.get(key, ()))
   645	                    if not spans.get(span_id, {}).get("text", "").strip():
   646	                        continue
   647	                    roles[span_id] = "page-varying" if variants > 1 else "boilerplate"
   648	            line["margin_span_roles"] = roles

   692	def _classify_paragraph_flow(pages: list[dict]) -> None:
   693	    """Mark whether each body line continues the previous line or starts a block."""
   694	    for page in pages:
   695	        body = [
   696	            line for line in page["lines"]
   697	            if not line.get("margin_band") and float(line["baseline_y"]) > 0
   698	            and line.get("ordered_span_ids")
   699	        ]
   700	        body.sort(key=lambda line: -float(line["baseline_y"]))
   701	        gaps = [
   702	            round(float(a["baseline_y"]) - float(b["baseline_y"]), 1)
   703	            for a, b in zip(body, body[1:])
   704	        ]
   705	        positive = sorted(gap for gap in gaps if gap > 0)
   706	        leading = positive[len(positive) // 2] if positive else 0.0
   707	        for line in page["lines"]:
   708	            line["flow"] = None
   709	            line["flow_gap"] = None
   710	        previous = None
   711	        for line in body:
   712	            if previous is None:
   713	                line["flow"] = "block-start"
   714	                previous = line
   715	                continue
   716	            gap = round(float(previous["baseline_y"]) - float(line["baseline_y"]), 1)
   717	            line["flow_gap"] = gap
   718	            same_block = (
   719	                leading > 0
   720	                and gap <= leading * 1.35
   721	                and line.get("bullet") is None
   722	                and line.get("indent_level") is not None
   723	                and line.get("indent_level") == previous.get("indent_level")
   724	                and previous["layout"]["kind"] == "single-flow"
   725	                and line["layout"]["kind"] == "single-flow"
   726	            )
   727	            line["flow"] = "wrap" if same_block else "block-start"
   728	            previous = line

   731	def _classify_page_columns(pages: list[dict]) -> None:
   732	    """Detect genuine side-by-side columns from disjoint horizontal extents."""
   733	    for page in pages:
   734	        body = [
   735	            line for line in page["lines"]
   736	            if not line.get("margin_band") and float(line["baseline_y"]) > 0
   737	            and line.get("ordered_span_ids")
   738	        ]
   739	        page["columns"] = []
   740	        for line in page["lines"]:
   741	            line["column_index"] = None
   742	        if len(body) < 6:
   743	            continue
   744	        extents = sorted(
   745	            ((float(line["x_range"][0]), float(line["x_range"][1]), index)
   746	             for index, line in enumerate(body)),
   747	        )
   748	        clusters = []
   749	        for left, right, index in extents:
   750	            line = body[index]
   751	            if clusters and left <= clusters[-1]["right"] + 36.0:
   752	                clusters[-1]["right"] = max(clusters[-1]["right"], right)
   753	                clusters[-1]["lines"].append(line)
   754	            else:
   755	                clusters.append({"left": left, "right": right, "lines": [line]})
   756	        if len(clusters) < 2:
   757	            continue
   758	        minimum_lines = max(5, int(len(body) * 0.25))
   759	        if any(len(cluster["lines"]) < minimum_lines for cluster in clusters):
   760	            continue
   761	        baselines = [
   762	            {round(float(line["baseline_y"]), 0) for line in cluster["lines"]}
   763	            for cluster in clusters
   764	        ]
   765	        shared = min(
   766	            len(left & right) / max(1, min(len(left), len(right)))
   767	            for left, right in zip(baselines, baselines[1:])
   768	        )
   769	        if shared < 0.5:
   770	            continue
   771	        for index, cluster in enumerate(clusters):
   772	            for line in cluster["lines"]:
   773	                line["column_index"] = index
   774	            page["columns"].append({
   775	                "index": index,
   776	                "x_range": [round(cluster["left"], 3), round(cluster["right"], 3)],
   777	                "line_count": len(cluster["lines"]),
   778	            })

   804	def _horizontal_span_order(line: dict, spans_by_id: dict[str, dict]) -> tuple[list[str], list[str]]:
   805	    """Return deterministic left-to-right evidence order for one baseline line."""
   806	    warnings = []
   807	    positioned = []
   808	    for span_id in line["span_ids"]:
   809	        span = spans_by_id[span_id]
   810	        position = span.get("position")
   811	        if position is None:
   812	            warnings.append(f"{line['id']}: {span_id}: missing-horizontal-position")
   813	            continue
   814	        positioned.append((float(position[0]), int(span.get("operation_index", 0)), span_id))
   815	    positioned.sort(key=lambda item: (item[0], item[1], item[2]))
   816	    for left, right in zip(positioned, positioned[1:]):
   817	        left_span = spans_by_id[left[2]]
   818	        right_span = spans_by_id[right[2]]
   819	        delta = abs(left[0] - right[0])
   820	        if (
   821	            0 < delta <= 0.25
   822	            and left_span.get("text", "").strip()
   823	            and right_span.get("text", "").strip()
   824	        ):
   825	            warnings.append(
   826	                f"{line['id']}: ambiguous-horizontal-order: {left[2]}, {right[2]}"
   827	            )
   828	    return [item[2] for item in positioned], warnings

   831	def _infer_span_separators(line: dict, spans_by_id: dict[str, dict]) -> None:
   832	    """Mark adjacent spans that need a separator in reconstructed text."""
   833	    ids = line.get("ordered_span_ids", [])
   834	    for span in (spans_by_id[span_id] for span_id in ids):
   835	        span["inferred_spacing"] = False
   836	    for left_id, right_id in zip(ids, ids[1:]):
   837	        left = spans_by_id[left_id]
   838	        right = spans_by_id[right_id]
   839	        left_text = left.get("text", "")
   840	        right_text = right.get("text", "")
   841	        if not left_text or not right_text:
   842	            continue
   843	        if left_text[-1].isspace() or right_text[0].isspace():
   844	            continue
   845	        if (left_text[-1].isalnum() or left_text[-1] in ":;,.]") and (
   846	            right_text[0].isalnum() or right_text[0] in "\"'([{•–—"
   847	        ):
   848	            right["inferred_spacing"] = True

   852	    """Join ordered spans using recorded geometric separator evidence."""

   862	def _cluster_spans_into_lines(spans: list[dict]) -> tuple[list[dict], list[str]]:
   863	    """Group positioned spans by baseline without imposing horizontal order."""
   864	    lines = []
   865	    warnings = []
   866	    for span in spans:
   867	        position = span.get("position")
   868	        if position is None:
   869	            warnings.append(f"{span['id']}: missing-position")
   870	            continue
   871	        x, y = position
   872	        font_size = max(float(span.get("font_size") or 0), 1.0)
   873	        tolerance = max(0.75, font_size * 0.25)
   874	        candidates = [line for line in lines if abs(line["baseline_y"] - y) <= max(line["tolerance"], tolerance)]
   875	        if candidates:
   876	            line = min(candidates, key=lambda item: (abs(item["baseline_y"] - y), item["operation_index"]))
   877	            count = len(line["span_ids"])
   878	            line["baseline_y"] = round((line["baseline_y"] * count + y) / (count + 1), 6)
   879	            line["tolerance"] = round(max(line["tolerance"], tolerance), 6)
   880	            line["span_ids"].append(span["id"] )
   881	            line["x_range"][0] = min(line["x_range"][0], x)
   882	            line["x_range"][1] = max(line["x_range"][1], x)
   883	        else:
   884	            lines.append({
   885	                "id": "",
   886	                "span_ids": [span["id"]],
   887	                "baseline_y": y,
   888	                "x_range": [x, x],
   889	                "tolerance": round(tolerance, 6),
   890	                "operation_index": int(span.get("operation_index", 0)),
   891	            })
   892	    lines.sort(key=lambda item: item["operation_index"])
   893	    for number, line in enumerate(lines, 1):
   894	        page = line["span_ids"][0].split("-s", 1)[0]
   895	        line["id"] = f"{page}-l{number}"
   896	        line["x_range"] = [round(value, 6) for value in line["x_range"]]
   897	    return lines, warnings

   900	def _span_orientation(text_matrix: list[float]) -> str:
   901	    """Classify span orientation from its text matrix."""
   902	    a, b, c, d = (float(value) for value in text_matrix[:4])
   903	    if abs(b) < 1e-6 and abs(c) < 1e-6:
   904	        if a > 0 and d > 0:
   905	            return "upright"
   906	        return "flipped"
   907	    if abs(a) < 1e-6 and abs(d) < 1e-6:
   908	        return "vertical"
   909	    return "skewed"

   921	def _pypdf_page_observations(path: Path) -> list[dict]:
   922	    """Capture raw pypdf text and geometry without changing text reconstruction.
   923	
   924	    This is deliberately an evidence-only first step: ``raw_text`` remains the
   925	    normal pypdf extraction result, while visitor fragments retain coordinates
   926	    for later, independently benchmarked line reconstruction.
   927	    """
   928	    from pypdf import PdfReader  # type: ignore
   929	
   930	    observations = []
   931	    for page_number, page in enumerate(PdfReader(str(path)).pages, 1):
   932	        spans = []
   933	
   934	        def visitor(text, current_matrix, text_matrix, font, font_size):
   935	            if not text:
   936	                return
   937	            spans.append({
   938	                "id": f"p{page_number}-s{len(spans) + 1}",
   939	                "text": text,
   940	                "current_matrix": _matrix_values(current_matrix),
   941	                "text_matrix": _matrix_values(text_matrix),
   942	                "position": _effective_text_position(current_matrix, text_matrix),
   943	                "orientation": _span_orientation(_matrix_values(text_matrix)),
   944	                "unmapped_glyphs": _unmapped_glyph_count(text),
   945	                "font": str((font or {}).get("/BaseFont", "")),
   946	                "font_size": round(float(font_size or 0), 6),
   947	                "operation_index": len(spans),
   948	                "inferred_spacing": False,
   949	                "inferred_line_break": False,
   950	            })
   951	
   952	        raw_text = page.extract_text(visitor_text=visitor) or ""
   953	        horizontal = [
   954	            span for span in spans
   955	            if span["orientation"] in ("upright", "flipped")
   956	            and not span["unmapped_glyphs"]
   957	        ]
   958	        lines, warnings = _cluster_spans_into_lines(horizontal)
   959	        for span in spans:
   960	            if span["orientation"] != "upright":
   961	                warnings.append({
   962	                    "kind": "non-upright-span",
   963	                    "span_id": span["id"],
   964	                    "orientation": span["orientation"],
   965	                })
   966	            if span["unmapped_glyphs"]:
   967	                warnings.append({
   968	                    "kind": "unmapped-glyphs",
   969	                    "span_id": span["id"],
   970	                    "count": span["unmapped_glyphs"],
   971	                    "font": span["font"],
   972	                })

   974	        for line in lines:
   975	            line["ordered_span_ids"], order_warnings = _horizontal_span_order(line, spans_by_id)
   976	            _infer_span_separators(line, spans_by_id)
   977	            same_origin_groups = []
   978	            current_group = []
   979	            current_x = None
   980	            for span_id in line["ordered_span_ids"]:
   981	                x = spans_by_id[span_id]["position"][0]
   982	                if current_x is not None and x != current_x:
   983	                    if len(current_group) > 1:
   984	                        same_origin_groups.append(current_group)
   985	                    current_group = []
   986	                current_group.append(span_id)
   987	                current_x = x
   988	            if len(current_group) > 1:
   989	                same_origin_groups.append(current_group)
   990	            line["same_origin_groups"] = same_origin_groups
   991	            line["layout"] = _classify_line_layout(line, spans_by_id)
   992	            warnings.extend(order_warnings)
   993	        _promote_repeated_cell_patterns(lines)
   994	        observations.append({
   995	            "page_number": page_number,
   996	            "raw_text": raw_text,
   997	            "spans": spans,
   998	            "lines": lines,
   999	            "warnings": warnings,
  1000	            "backend": "pypdf",
  1001	        })
  1002	    _classify_repeated_margin_bands(observations)
  1003	    _classify_list_structure(observations)
  1004	    _classify_paragraph_flow(observations)
  1005	    _classify_page_columns(observations)
  1006	    _finalize_line_order(observations)
  1007	    return observations

  1124	def classify_page_structure(text: str) -> dict:
  1125	    """Label the non-body regions of a page once, for reuse across phases.
  1126	
  1127	    Extraction, segmentation and the benchmark all need to know which parts of
  1128	    a page are appendix history, table of contents or bibliography.  Detecting
  1129	    that separately in each phase invites drift, so the labels are produced
  1130	    here and consumed everywhere else.
  1131	    """
  1132	    regions = [{"kind": "history", "start": start, "stop": stop}
  1133	               for start, stop in _history_regions(text)]
  1134	    regions.extend({"kind": "traceability", "start": start, "stop": stop}
  1135	                   for start, stop in _traceability_regions(text))
  1136	    for match in TOC_LINE_RE.finditer(text):
  1137	        regions.append({"kind": "toc", "start": match.start(), "stop": match.end()})
  1138	    for match in BIBLIO_HEADING_RE.finditer(text):
  1139	        regions.append({"kind": "bibliography", "start": match.start(),
  1140	                        "stop": min(len(text), match.end() + 4000)})
  1141	    regions.sort(key=lambda item: (item["kind"], item["start"], item["stop"]))
  1142	    merged = []
  1143	    for item in regions:
  1144	        if merged and merged[-1]["kind"] == item["kind"] and item["start"] <= merged[-1]["stop"]:
  1145	            merged[-1]["stop"] = max(merged[-1]["stop"], item["stop"])
  1146	            continue
  1147	        merged.append(dict(item))
  1148	    regions = sorted(merged, key=lambda item: (item["start"], item["stop"]))
  1149	    return {"regions": regions,
  1150	            "kinds": sorted({item["kind"] for item in regions})}

  1161	def _history_regions(text: str) -> list:
  1162	    """Character ranges of appendix history tables and change-log blocks.
  1163	
  1164	    A page may hold both a history table and ordinary body text, so rejection
  1165	    has to work per occurrence.  A region starts at a history heading and ends
  1166	    at the caption of the table it introduces, or at the next history heading,
  1167	    or after a bounded window when neither is present.
  1168	    """
  1169	    regions = []
  1170	    for match in HISTORY_HEADING_RE.finditer(text):
  1171	        caption = HISTORY_TABLE_CAPTION_RE.search(text, match.end())
  1172	        nxt = HISTORY_HEADING_RE.search(text, match.end())
  1173	        stop = min([x for x in (caption.end() if caption else None,
  1174	                                nxt.start() if nxt else None,
  1175	                                match.end() + 4000) if x is not None])
  1176	        regions.append((match.start(), stop))
  1177	    return regions

  1257	def _ids_for_document(task: tuple[str, str, str | None, frozenset, bool]) -> tuple[str, dict]:
  1258	    """Per-document worker for phase_ids.
  1259	
  1260	    Each source PDF is independent, so ID discovery is naturally parallel at the
  1261	    document level. The old implementation scanned every PDF sequentially,
  1262	    calling ``pdf_pages`` and then running all regex/history heuristics in a
  1263	    single process; that kept one core saturated and blocked later parallelized
  1264	    phases from ever starting.
  1265	    """
  1266	    path_str, backend, pattern, keep, include_refs = task
  1267	    path = Path(path_str)
  1268	    started = time.perf_counter()
  1269	    rx = re.compile(pattern, re.IGNORECASE) if pattern else None
  1270	    pages = [strip_noise(x) for x in pdf_pages(path, backend)]
  1271	    hits = defaultdict(list)
  1272	    spellings = defaultdict(list)
  1273	    in_history_continuation = False
  1274	    for pageno, text in enumerate(pages, 1):
  1275	        continuation = in_history_continuation and _history_continuation_page(text)
  1276	        definiert = set() if continuation else _definition_ids(text)
  1277	        for raw_rid in ID_RE.findall(text):
  1278	            rid = raw_rid.upper()
  1279	            if rx and not rx.search(rid):
  1280	                continue
  1281	            if keep and rid not in keep:
  1282	                continue
  1283	            if not include_refs and rid not in definiert:
  1284	                continue
  1285	            if pageno not in hits[rid]:
  1286	                hits[rid].append(pageno)
  1287	            if raw_rid not in spellings[rid]:
  1288	                spellings[rid].append(raw_rid)
  1289	        regions = _history_regions(text)
  1290	        starts_new_run = bool(regions and any(stop >= len(text) - 5 for _, stop in regions))
  1291	        in_history_continuation = continuation or starts_new_run
  1292	    evidence = _history_only_evidence(pages) if not include_refs else {}
  1293	    rejected = sorted(rid for rid in evidence if not rx or rx.search(rid))
  1294	    for rid in rejected:
  1295	        hits.pop(rid, None)
  1296	        spellings.pop(rid, None)
  1297	    doc_result = {"path": str(path), "pages": len(pages),
  1298	                  "ids": {k: v for k, v in sorted(hits.items())},
  1299	                  "spellings": {k: v for k, v in sorted(spellings.items())},
  1300	                  "history_only_ids": list(rejected),
  1301	                  "history_only_evidence": {rid: evidence[rid] for rid in rejected}}
  1302	    elapsed = time.perf_counter() - started
  1303	    LOG.info("phase_ids document done name=%s ids=%d elapsed=%.3fs", path.name, len(doc_result["ids"]), elapsed)
  1304	    return path.name, doc_result

  1307	def phase_ids(pdfs, pattern=None, only_ids=None, include_refs=False,
  1308	              backend="auto", jobs: int | None = None) -> dict:
  1309	    """-> {pdf-name: {id: [seitenzahlen]}}"""
  1310	    keep = frozenset(rid.upper() for rid in (only_ids or ()))
  1311	    paths = [str(path) for path in pdfs]
  1312	    result = OrderedDict()
  1313	    if not paths:
  1314	        return result
  1315	    worker_count = max(1, min(len(paths), jobs or (os.cpu_count() or 1)))
  1316	    started = time.perf_counter()
  1317	    LOG.info("phase_ids start documents=%d workers=%d", len(paths), worker_count)
  1318	    tasks = [(path_str, backend, pattern, keep, include_refs) for path_str in paths]
  1319	    if worker_count == 1:
  1320	        results = [_ids_for_document(task) for task in tasks]
  1321	    else:
  1322	        with ProcessPoolExecutor(max_workers=worker_count) as executor:
  1323	            results = list(executor.map(_ids_for_document, tasks, chunksize=1))
  1324	    by_name = dict(results)
  1325	    for path_str in paths:
  1326	        path = Path(path_str)
  1327	        result[path.name] = by_name[path.name]
  1328	    elapsed = time.perf_counter() - started
  1329	    total_ids = sum(len(info.get("ids", {})) for info in result.values())
  1330	    LOG.info("phase_ids done documents=%d ids=%d elapsed=%.3fs rate=%.1f docs/s", len(paths), total_ids, elapsed, len(paths) / max(elapsed, 1e-9))
  1331	    return result

  1344	    eine Inline-Zitierung derselben ID (z. B. in ``Dependencies``). Sonst kann

  1346	    gesuchten ID zuordnen.
  1347	
  1348	    Grenze ist entweder das Spec-Item-Ende (⌋) oder die naechste in eckigen
  1349	    Klammern stehende ID — sonst laufen die Eigenschaften des Folgerecords in
  1350	    den aktuellen hinein.
  1351	    """
  1352	    pattern = re.compile(r"\[%s\]" % re.escape(rid), re.IGNORECASE)
  1353	    matches = list(pattern.finditer(text))
  1354	    m = None
  1355	    for cand in matches:
  1356	        lookahead = text[cand.end():cand.end() + 240]
  1357	        if "⌈" in lookahead:
  1358	            pos = lookahead.find("⌈")
  1359	            nxt = DEF_RE.search(lookahead)
  1360	            if nxt is None or nxt.start() > pos:
  1361	                m = cand
  1362	                break
  1363	    if m is None and matches:
  1364	        m = matches[0]
  1365	    if not m:
  1366	        m = re.search(re.escape(rid), text, re.IGNORECASE)
  1367	        if not m:
  1368	            return ""
  1369	    rest = text[m.end():]
  1370	    grenzen = []
  1371	    ende = rest.find("⌋")
  1372	    if ende >= 0:
  1373	        grenzen.append(ende)
  1374	    nxt = DEF_RE.search(rest)
  1375	    if nxt:
  1376	        grenzen.append(nxt.start())
  1377	    return rest[:min(grenzen)] if grenzen else rest[:6000]

  1428	def fix_ligatures(text: str) -> str:
  1429	    """Von pypdf zerrissene Bezeichner wieder zusammenfuegen.
  1430	
  1431	    pypdf trennt Ligaturen ("RS_L T_00003", "Log and T race"). Innerhalb von
  1432	    Spec-IDs ist ein Leerzeichen nie gueltig, daher kann es dort gefahrlos
  1433	    entfernt werden.
  1434	    """
  1435	    text = LIGATUR_RE.sub(lambda m: "%s_%s_%s" % (m.group(1),
  1436	                                                  m.group(2).replace(" ", ""),
  1437	                                                  m.group(3)), text or "")
  1438	    # "T race" -> "Trace", "T emplate" -> "Template": Grossbuchstabe, Leerzeichen,
  1439	    # Kleinbuchstabenrest. Nur bei bekannten Woertern, um echte Wortgrenzen
  1440	    # nicht zu zerstoeren.
  1441	    return re.sub(r"\b([A-Z]) (race|emplate|ype|ime|hread|able)\b",
  1442	                  lambda m: m.group(1) + m.group(2), text)

  1445	def strip_noise(text: str) -> str:
  1446	    """Seitenkopf und -fuss entfernen.
  1447	
  1448	    Die Eigenschaftstabellen laufen ueber Seitengrenzen; ohne diese Bereinigung
  1449	    landet die Fusszeile mitten im Zellwert ("std::uint8_t ▽ 52 of 122 Document
  1450	    ID 853: ...").
  1451	    """
  1452	    text = fix_ligatures(text)
  1453	    for _ in range(2):            # Titel kann durch Umbruch zweigeteilt sein
  1454	        for rx in NOISE_RES:
  1455	            text = rx.sub(" ", text)
  1456	        text = fix_ligatures(text)
  1457	    # Alleinstehende Ligaturreste einer bereits entfernten Fusszeile.
  1458	    text = re.sub(r"(?m)^[ \t]*(race|emplate|ime|hread|able|ype)[ \t]*$", " ", text)
  1459	    return text

  1470	def normalize_layout(text: str) -> str:
  1471	    """Zeilenstruktur erzwingen.
  1472	
  1473	    Je nach Backend (pypdf, PyMuPDF, eingebauter Extraktor) kommen die
  1474	    Tabellenzellen mit, ohne oder mit falsch gesetzten Zeilenumbruechen an.
  1475	    Deshalb wird vor jeder bekannten Beschriftung und vor jeder in eckigen
  1476	    Klammern stehenden ID hart umbrochen — danach ist die Auswertung
  1477	    backend-unabhaengig.
  1478	    """
  1479	    text = re.sub(r"[ \t]+", " ", strip_noise(text))
  1480	    text = DEF_RE.sub(lambda m: "\n[%s]\n" % m.group(1), text)
  1481	    return NORM_RE.sub("\n", text)

  1484	def _requirement_text(chunk: str) -> str | None:
  1485	    """Normativen Freitext eines Spec-Items zwischen ⌈ und ⌋ extrahieren.
  1486	
  1487	    API-Records enthalten in diesem Bereich Eigenschaftstabellen; für sie wird
  1488	    kein ``requirement_text`` zurückgegeben. Prosa-Anforderungen haben dagegen
  1489	    genau hier ihren kanonischen SHALL/SHOULD/MAY-Text. Die Unterscheidung über
  1490	    eine bekannte Tabellenbeschriftung hält beide Record-Arten sauber getrennt.
  1491	    """
  1492	    if "⌈" not in chunk:
  1493	        return None
  1494	    text = chunk.split("⌈", 1)[1]
  1495	    if "⌋" in text:
  1496	        text = text.split("⌋", 1)[0]
  1497	    lines = [line.strip() for line in text.split("\n") if line.strip()]
  1498	    if any(LABEL_RE.match(line) for line in lines):
  1499	        return None
  1500	    value = _clean_value(" ".join(lines))
  1501	    return value or None

  1504	def _subsection_heading_before(text: str, rid: str) -> str | None:
  1505	    """Fallback-Ueberschrift aus der nummerierten Zeile vor ``[rid]``.
  1506	
  1507	    Manche Dokumente (z. B. AUTOSAR_FO_RS_LogAndTrace) setzen den Titeltext
  1508	    nicht in den Record selbst, sondern in die nummerierte Unterabschnitts-
  1509	    Zeile direkt davor; der [ID]-Marker folgt dort unmittelbar von der
  1510	    oeffnenden Klammer ohne eigenen Titeltext. ``text`` ist bereits durch
  1511	    ``normalize_layout`` auf harte Zeilenumbrueche vor jeder [ID] normiert.
  1512	    Der Unterabschnittstitel kann ueber mehrere physische Zeilen umbrechen, die
  1513	    erste Zeile traegt aber stets das Nummernpraefix ``4.2.1...``.
  1514	    """
  1515	    pattern = re.compile(r"\[%s\]" % re.escape(rid), re.IGNORECASE)
  1516	    m = pattern.search(text)
  1517	    if not m:
  1518	        return None
  1519	    lines = text[:m.start()].split("\n")
  1520	    tail = []
  1521	    seen_text = False
  1522	    for line in reversed(lines):
  1523	        line = line.strip()
  1524	        if not line:
  1525	            if seen_text:
  1526	                break
  1527	            continue
  1528	        seen_text = True
  1529	        tail.append(line)
  1530	        sub = SUBSECTION_HEADING_RE.match(line)
  1531	        if sub:
  1532	            prefix = sub.group(1)
  1533	            rest = " ".join(reversed(tail[:-1]))
  1534	            return _clean_value((prefix + " " + rest).strip())[:120] or None
  1535	    return None

  1538	def parse_record(text: str, rid: str) -> dict:
  1539	    """Eigenschaften eines Spec-Records aus dem PDF-Text."""
  1540	    chunk = _record_slice(normalize_layout(text), rid)
  1541	    rec = {"id": rid, "props": OrderedDict(), "upstream": [], "heading": None,
  1542	           "requirement_text": None}
  1543	    if not chunk:
  1544	        return rec
  1545	    rec["requirement_text"] = _requirement_text(chunk)
  1546	    for ups in UPSTREAM_RE.finditer(chunk):
  1547	        for uid in ID_RE.findall(ups.group(1).split("\n")[0]):
  1548	            uid = uid.upper()
  1549	            if uid not in rec["upstream"]:
  1550	                rec["upstream"].append(uid)
  1551	    # A heading spans everything between the ID and the Status/table marker.
  1552	    # builtin deliberately emits more positioning newlines than pypdf, so using
  1553	    # only the first physical line truncates headings such as UCM to one word.
  1554	    head_part = re.split(r"(?:⌈|(?:^|\n)\s*(?:Status|Upstream requirements?|Kind)\s*:?)",
  1555	                         chunk.lstrip("\n"), maxsplit=1)[0]
  1556	    head = _clean_value(" ".join(line.strip() for line in head_part.split("\n")
  1557	                                  if line.strip()))
  1558	    if head and not HEADING_LABEL_RE.match(head):
  1559	        rec["heading"] = head[:120]
  1560	    if rec["heading"] is None:
  1561	        rec["heading"] = _subsection_heading_before(normalize_layout(text), rid)
  1562	    current, buf = None, []
  1563	    for line in chunk.split("\n"):
  1564	        line = line.strip()
  1565	        if not line:
  1566	            continue
  1567	        m = LABEL_RE.match(line)
  1568	        if m:
  1569	            if current:
  1570	                rec["props"][current] = _clean_value(" ".join(buf))
  1571	            current, buf = m.group(1), ([m.group(2).strip()] if m.group(2).strip() else [])
  1572	        elif current:
  1573	            buf.append(line)
  1574	    if current:
  1575	        rec["props"][current] = _clean_value(" ".join(buf))
  1576	    ns, enclosing = namespace_from_scope(rec["props"].get("Scope", ""))
  1577	    rec["namespace"], rec["enclosing"] = ns, enclosing
  1578	    return rec

  1581	def namespace_from_scope(scope: str):
  1582	    """'namespace ara::log' -> ('ara::log', None);
  1583	    'class ara::log::LogStream' -> ('ara::log', 'ara::log::LogStream').
  1584	
  1585	    Die Ableitung beruht auf einem Whitelist-Match ueber bekannte
  1586	    Namespace-Praefixe. Gross-/Kleinschreibung ist dafuer ungeeignet, weil es
  1587	    kleingeschriebene Typen gibt (z. B. std::hash, value_compare, reference,
  1588	    in_place_t), die keine Namespaces sind.
  1589	    """
  1590	    original = (scope or "").strip()
  1591	    s = re.sub(r"^(namespace|class|struct|enum(?:\s+class)?|union)\s+", "", original).strip()
  1592	    while re.search(r"<[^<>]*>", s):
  1593	        s = re.sub(r"<[^<>]*>", "", s)
  1594	    s = re.sub(r"\s+", " ", s).strip()
  1595	    if not s:
  1596	        return None, None
  1597	    if original.startswith("namespace"):
  1598	        return s, None
  1599	    for prefix in KNOWN_NAMESPACE_PREFIXES:
  1600	        if s == prefix:
  1601	            return prefix, None
  1602	        if s.startswith(prefix + "::"):
  1603	            return prefix, s
  1604	    parts = [p for p in (seg.strip() for seg in s.split("::")) if p]
  1605	    if not parts or parts[0] not in ("ara", "apext", "std"):
  1606	        return None, None
  1607	    if len(parts) == 1:
  1608	        return parts[0], None
  1609	    return parts[0], s

  1615	def _record_page_span(pages: list, start_page: int, rid: str) -> tuple[list[int], bool]:
  1616	    """Inclusive page numbers a record occupies, plus whether its end was seen.
  1617	
  1618	    The window grows until ``_record_slice`` observes a terminator (spec-item
  1619	    end or the next bracketed definition).  Without a terminator the record is
  1620	    reported as unterminated instead of silently claiming a single page.
  1621	    """
  1622	    last = len(pages)
  1623	    for width in range(1, MAX_RECORD_PAGES + 1):
  1624	        stop = min(start_page - 1 + width, last)
  1625	        joined = "\n".join(pages[start_page - 1:stop])
  1626	        m = re.search(re.escape(rid), joined, re.IGNORECASE)
  1627	        if not m:
  1628	            continue
  1629	        rest = joined[m.end():]
  1630	        if "\u230b" in rest or DEF_RE.search(rest):
  1631	            consumed = rest.split("\u230b")[0]
  1632	            nxt = DEF_RE.search(rest)
  1633	            if nxt:
  1634	                consumed = min(consumed, rest[:nxt.start()], key=len)
  1635	            # Count how many of the joined pages the consumed slice reaches into.
  1636	            offset = m.end() + len(consumed)
  1637	            used, cursor = start_page, 0
  1638	            for idx in range(start_page - 1, stop):
  1639	                cursor += len(pages[idx]) + 1
  1640	                used = idx + 1
  1641	                if cursor >= offset:
  1642	                    break
  1643	            return list(range(start_page, used + 1)), True
  1644	        if stop >= last:
  1645	            break
  1646	    return list(range(start_page, min(start_page, len(pages)) + 1)), False

  1649	def _props_for_document(task: tuple[str, dict, frozenset, str]) -> tuple[str, dict]:
  1650	    """Per-document worker: extract+parse one PDF's records. Independent of
  1651	    every other document, so this is the natural unit of parallelism --
  1652	    ``phase_props`` previously ran this loop body sequentially for every
  1653	    document, which meant only one core was ever busy on PDF extraction and
  1654	    regex-heavy parsing across ~7500+ IDs.
  1655	    """
  1656	    name, info, keep, backend = task
  1657	    started = time.perf_counter()
  1658	    pages = [strip_noise(x) for x in pdf_pages(Path(info["path"]), backend)]
  1659	    doc_out: dict = {}
  1660	    for rid, pagenos in info["ids"].items():
  1661	        if keep and rid not in keep:
  1662	            continue
  1663	        best = None
  1664	        for pageno in pagenos:
  1665	            joined = "\n".join(pages[pageno - 1 : pageno + 1])  # Seitenumbruch mitnehmen
  1666	            rec = parse_record(joined, rid)
  1667	            if rec["props"] or rec["upstream"]:
  1668	                best = rec
  1669	                break
  1670	        rec = best or {"id": rid, "props": {}, "upstream": [], "heading": None,
  1671	                      "requirement_text": None, "namespace": None, "enclosing": None}
  1672	        rec["document"] = name
  1673	        rec["page"] = pagenos[0] if pagenos else None
  1674	        if pagenos:
  1675	            span, terminated = _record_page_span(pages, pagenos[0], rid)
  1676	        else:
  1677	            span, terminated = [], False
  1678	        rec["pages"] = span
  1679	        rec["pages_all_definitions"] = list(pagenos)
  1680	        rec["complete_end"] = terminated
  1681	        rec["id_observed"] = (info.get("spellings", {}).get(rid) or [rid])[0]
  1682	        doc_out[rid] = rec
  1683	    elapsed = time.perf_counter() - started
  1684	    LOG.info(
  1685	        "phase_props document done name=%s ids=%d elapsed=%.3fs",
  1686	        name, len(doc_out), elapsed,
  1687	    )
  1688	    return name, doc_out

  1691	def phase_props(index: dict, only_ids=None, backend="auto", jobs: int | None = None) -> dict:
  1692	    keep = frozenset(only_ids or ())
  1693	    out = OrderedDict()
  1694	    documents = list(index.items())
  1695	    if not documents:
  1696	        return out
  1697	    worker_count = max(1, min(len(documents), jobs or (os.cpu_count() or 1)))
  1698	    started = time.perf_counter()
  1699	    LOG.info("phase_props start documents=%d workers=%d", len(documents), worker_count)
  1700	    tasks = [(name, info, keep, backend) for name, info in documents]
  1701	    if worker_count == 1:
  1702	        results = [_props_for_document(task) for task in tasks]
  1703	    else:
  1704	        with ProcessPoolExecutor(max_workers=worker_count) as executor:
  1705	            results = list(executor.map(_props_for_document, tasks, chunksize=1))
  1706	    # Preserve original document-then-id insertion order regardless of the
  1707	    # (nondeterministic) completion order of parallel workers.
  1708	    by_name = dict(results)
  1709	    for name, _info in documents:
  1710	        out.update(by_name.get(name, {}))
  1711	    elapsed = time.perf_counter() - started
  1712	    LOG.info(
  1713	        "phase_props done documents=%d ids=%d elapsed=%.3fs rate=%.1f docs/s",
  1714	        len(documents), len(out), elapsed, len(documents) / max(elapsed, 1e-9),
  1715	    )
  1716	    return out

  1718	def _repair_requirements(requirements: dict, pages_by_doc: dict,
  1719	                         agreement: dict | None = None) -> None:
  1720	    """Normtext reparieren, Rohtext erhalten, Evidenzlage bewerten.
  1721	
  1722	    Der Lexikon-Guard stammt aus dem jeweiligen PDF selbst; damit ist jede
  1723	    Reparatur durch das Quelldokument belegt und nicht durch externes Wissen.

  1725	    import text_repair
  1726	    identifiers = _known_identifiers()

  1728	                for doc, pages in pages_by_doc.items()}
  1729	    for rid, rec in requirements.items():
  1730	        lex = lexicons.get(rec.get("document")) or {}
  1731	        entry = text_repair.repair_text(rec.get("requirement_text") or "",
  1732	                                        lex, identifiers)
  1733	        agree = (agreement or {}).get(rid)
  1734	        conf, review, reason = text_repair.assess(entry, agree)
  1735	        rec["requirement_text"] = entry["text_en"]
  1736	        rec["requirement_text_raw"] = entry["text_raw"]
  1737	        rec["repairs"] = entry["repairs"]
  1738	        rec["suspects"] = entry["suspects"]
  1739	        rec["confidence"] = conf

  1741	        rec["review_reason"] = reason

  1744	def _known_identifiers() -> set:
  1745	    """Bekannte C++-Bezeichner aus der Spec-DB als CamelCase-Whitelist."""
  1746	    names = set()
  1747	    if not RECORDS.exists():
  1748	        return names
  1749	    ident = re.compile(r"[A-Za-z_][A-Za-z0-9_]*")
  1750	    for path in RECORDS.glob("*/*.json"):
  1751	        try:
  1752	            raw = path.read_text(encoding="utf-8")
  1753	        except OSError:
  1754	            continue
  1755	        for tok in ident.findall(raw):
  1756	            if re.search(r"[a-z][A-Z]", tok):
  1757	                names.add(tok.lower())
  1758	    return names

  1917	def phase_compare(scraped: dict, prefixes=None, rebuild=False, vollstaendig=True) -> dict:
  1918	    db = load_db(prefixes)
  1919	    report = {"checked": 0, "only_in_pdf": [], "only_in_db": [], "diffs": [],
  1920	              "namespace_diffs": [], "enclosing_diffs": [],
  1921	              "namespace_legacy_schema": [], "empty_extraction": [], "written": []}
  1922	    for rid, rec in scraped.items():
  1923	        if not rec["props"]:
  1924	            report["empty_extraction"].append(rid)
  1925	        if rid not in db:
  1926	            report["only_in_pdf"].append(rid)
  1927	            continue
  1928	        report["checked"] += 1
  1929	        entry = db[rid]
  1930	        for label in COMPARED:
  1931	            pdf_val = rec["props"].get(label)
  1932	            db_val = entry["kind"] if label == "Kind" else entry["props"].get(label)
  1933	            if pdf_val and db_val and _norm(pdf_val) != _norm(db_val):
  1934	                report["diffs"].append({"id": rid, "field": label,
  1935	                                        "pdf": pdf_val[:120], "db": db_val[:120]})
  1936	            elif pdf_val and not db_val:
  1937	                report["diffs"].append({"id": rid, "field": label,
  1938	                                        "pdf": pdf_val[:120], "db": None})
  1939	        # Namespace und umschliessender Typ sind zwei verschiedene Fakten und
  1940	        # werden getrennt verglichen. Wo die DB noch dem Altschema folgt und
  1941	        # den umschliessenden Typ im Feld ``namespace`` fuehrt, ist das kein
  1942	        # Datenfehler des Inhalts, sondern ein Schemarest: eigener Topf,
  1943	        # damit nichts still verschwindet (siehe SPEC_BUILD_PROCESS.md).
  1944	        pdf_ns, pdf_enc = rec.get("namespace"), rec.get("enclosing")
  1945	        meta = entry.get("namespace_meta") or {}
  1946	        db_ns, db_enc = meta.get("namespace"), meta.get("enclosing")
  1947	        if pdf_ns and db_ns and _norm(pdf_ns) != _norm(db_ns):
  1948	            if db_enc is None and pdf_enc and _norm(db_ns) == _norm(pdf_enc):
  1949	                report["namespace_legacy_schema"].append(
  1950	                    {"id": rid, "namespace": pdf_ns, "enclosing": pdf_enc,
  1951	                     "db": db_ns, "rule": "enclosing-in-namespace"})
  1952	            elif db_enc is None and pdf_enc and _norm(pdf_enc).startswith(_norm(db_ns) + "::"):
  1953	                report["namespace_legacy_schema"].append(
  1954	                    {"id": rid, "namespace": pdf_ns, "enclosing": pdf_enc,
  1955	                     "db": db_ns, "rule": "enclosing-prefix-in-namespace"})
  1956	            else:
  1957	                report["namespace_diffs"].append({"id": rid, "pdf": pdf_ns, "db": db_ns})
  1958	        if pdf_enc and db_enc and _norm(pdf_enc) != _norm(db_enc):
  1959	            report["enclosing_diffs"].append({"id": rid, "pdf": pdf_enc, "db": db_enc})
  1960	    # "nur in der DB" ist ausschliesslich bei einem vollstaendigen Dokumentlauf
  1961	    # eine echte Aussage — bei --id/--limit/--pattern waere die Liste blosses
  1962	    # Rauschen (alles, was der Teillauf nicht angefasst hat).
  1963	    if vollstaendig and scraped:
  1964	        report["only_in_db"] = sorted(set(db) - set(scraped))
  1965	    else:
  1966	        report["only_in_db_unterdrueckt"] = True
  1967	    if rebuild:
  1968	        report["written"] = _rebuild(scraped, db)
  1969	    return report

  1972	def _rebuild(scraped: dict, db: dict) -> list:
  1973	    """Autoritative PDF-Felder in die DB zurueckschreiben.
  1974	
  1975	    Konservativ: nur der ``ns``-Block und ein ``quelle``-Vermerk werden
  1976	    aktualisiert bzw. ergaenzt. Darstellungsbloecke (HTML, KI-Fragmente,
  1977	    Property-Zeilen) bleiben unberuehrt — sie werden vom Generator gebraucht.
  1978	    """
  1979	    written = []
  1980	    for rid, rec in scraped.items():
  1981	        entry = db.get(rid)
  1982	        if not entry or not rec.get("namespace"):
  1983	            continue
  1984	        data = json.loads(entry["path"].read_text(encoding="utf-8"))
  1985	        ns = dict(data.get("namespace_meta") or data.get("ns") or {})
  1986	        before = dict(ns)
  1987	        ns["namespace"] = rec["namespace"]
  1988	        if rec.get("enclosing"):
  1989	            ns["enclosing"] = rec["enclosing"]
  1990	        ns.setdefault("module", (entry.get("namespace_meta") or {}).get("module") or (entry.get("namespace_meta") or {}).get("modul"))
  1991	        ns["source"] = "pdf"
  1992	        if rec.get("enclosing"):
  1993	            ns["umschliessend"] = rec["enclosing"]
  1994	        if ns == before:
  1995	            continue
  1996	        data["ns"] = ns
  1997	        entry["path"].write_text(json.dumps(data, ensure_ascii=False, indent=1) + "\n",
  1998	                                 encoding="utf-8")
  1999	        written.append(rid)
  2000	    return written

  2006	def compare_backend_results(results: dict) -> list:
  2007	    """Feldgenauer Vergleich: IDs, Metadaten, Upstream und alle PDF-Properties."""
  2008	    names = list(results)
  2009	    if len(names) < 2:
  2010	        return []
  2011	    left, right = names[0], names[1]
  2012	    a, b = results[left], results[right]
  2013	    deviations = []
  2014	    for rid in sorted(set(a) | set(b)):
  2015	        if rid not in a or rid not in b:
  2016	            deviations.append({"id": rid, "field": "record",
  2017	                               left: "present" if rid in a else "missing",
  2018	                               right: "present" if rid in b else "missing"})
  2019	            continue
  2020	        fields = {"heading", "namespace", "enclosing", "document", "page"}
  2021	        for field in sorted(fields):
  2022	            av, bv = a[rid].get(field), b[rid].get(field)
  2023	            if _norm(str(av or "")) != _norm(str(bv or "")):
  2024	                deviations.append({"id": rid, "field": field, left: av, right: bv})
  2025	        au, bu = sorted(a[rid].get("upstream") or []), sorted(b[rid].get("upstream") or [])
  2026	        if au != bu:
  2027	            deviations.append({"id": rid, "field": "upstream", left: au, right: bu})
  2028	        for field in sorted(set(a[rid].get("props", {})) | set(b[rid].get("props", {}))):
  2029	            av = a[rid].get("props", {}).get(field)
  2030	            bv = b[rid].get("props", {}).get(field)
  2031	            if _norm(str(av or "")) != _norm(str(bv or "")):
  2032	                deviations.append({"id": rid, "field": "props." + field,
  2033	                                   left: av, right: bv})
  2034	    return deviations

  2335	    record -> upstream RS -> every other record that satisfies the same RS,

  2414	def check_traceability_consistency(trace_index: dict | None = None) -> dict:
  2415	    """Bestehende Spec-Records gegen Kapitel-6-Traceability-Records abgleichen.
  2416	
  2417	    Zwei Widerspruchsarten werden erkannt:
  2418	
  2419	    - ``upstream_not_traced``: ein Record nennt ein RS als Upstream, aber die
  2420	      Traceability-Tabelle dieses RS listet den Record nicht unter
  2421	      "Satisfied by" (oder das RS taucht in der Traceability-DB gar nicht auf).
  2422	    - ``traced_not_upstream``: die Traceability-Tabelle nennt einen Record als
  2423	      Erfüller eines RS, der Record selbst weist dieses RS aber nicht (oder
  2424	      ein anderes) als Upstream aus.
  2425	
  2426	    Beide Fälle werden als Review-Kandidaten zurückgegeben, nicht automatisch
  2427	    verändert — die Kuratierung bleibt bewusst ein manueller Schritt.
  2428	    """
  2429	    def upstream_provenance(rec: dict, path: Path, rs_id: str) -> dict:
  2430	        upstream = []
  2431	        for item in (rec.get("upstream") or []):
  2432	            if not isinstance(item, dict):
  2433	                continue
  2434	            current_id = str(item.get("id", "")).upper()
  2435	            entry = {k: item[k] for k in ("id", "document", "page", "url") if item.get(k) is not None}
  2436	            entry["id"] = current_id
  2437	            upstream.append(entry)
  2438	        matched = [u for u in upstream if u.get("id") == rs_id]
  2439	        return {
  2440	            "origin": "record_upstream_field",
  2441	            "record_path": str(path),
  2442	            "record_id": str(rec.get("id", "")).upper(),
  2443	            "matched_upstream": matched,
  2444	            "all_upstream": upstream,
  2445	        }
  2446	
  2447	    def trace_side_provenance(trace_row: dict, rs_id: str, satisfier: str | None = None) -> dict:
  2448	        source_rows = []
  2449	        for row in ((trace_row.get("provenance") or {}).get("source_rows") or []):
  2450	            if satisfier is None or satisfier in set(row.get("satisfied_by") or []):
  2451	                source_rows.append(row)
  2452	        return {
  2453	            "origin": (trace_row.get("provenance") or {}).get("origin") or "chapter6_requirements_tracing",
  2454	            "record_path": trace_row.get("path"),
  2455	            "record_id": rs_id,
  2456	            "document": trace_row.get("document"),
  2457	            "page": trace_row.get("page"),
  2458	            "description": trace_row.get("description") or "",
  2459	            "source_documents": list(((trace_row.get("provenance") or {}).get("source_documents") or [])),
  2460	            "matched_source_rows": source_rows,
  2461	        }
  2462	
  2463	    trace_index = trace_index if trace_index is not None else load_traceability_index()
  2464	    flagged = []
  2465	    checked = 0
  2466	    for path in sorted(RECORDS.rglob("*.json")):
  2467	        try:
  2468	            rec = json.loads(path.read_text(encoding="utf-8"))
  2469	        except (json.JSONDecodeError, OSError):
  2470	            continue
  2471	        rid = str(rec.get("id", "")).upper()
  2472	        if not rid:
  2473	            continue
  2474	        upstream_ids = [str(u.get("id", "")).upper() for u in (rec.get("upstream") or [])
  2475	                        if isinstance(u, dict) and u.get("id")]
  2476	        if not upstream_ids:
  2477	            continue
  2478	        checked += 1
  2479	        for rs_id in upstream_ids:
  2480	            trace_row = trace_index.get(rs_id)
  2481	            if trace_row is None:
  2482	                flagged.append({
  2483	                    "record": rid, "path": str(path), "rs_id": rs_id,
  2484	                    "issue": "upstream_not_traced",
  2485	                    "detail": "RS %s hat keinen Traceability-Record." % rs_id,
  2486	                    "provenance": {
  2487	                        "db_upstream": upstream_provenance(rec, path, rs_id),
  2488	                        "traceability": {
  2489	                            "origin": "chapter6_requirements_tracing",
  2490	                            "record_path": None,
  2491	                            "record_id": rs_id,
  2492	                            "matched": False,
  2493	                            "reason": "missing_traceability_record",
  2494	                            "matched_source_rows": [],
  2495	                            "source_documents": [],
  2496	                        },
  2497	                    },
  2498	                })
  2499	            elif rid not in trace_row["satisfied_by"]:
  2500	                satisfied_by = trace_row["satisfied_by"]
  2501	                if not satisfied_by:
  2502	                    reason = "traceability_row_has_empty_satisfied_by"
  2503	                    detail = "RS %s hat einen Traceability-Record, dessen 'Satisfied by'-Liste leer ist." % rs_id
  2504	                else:
  2505	                    reason = "record_not_listed_under_satisfied_by"
  2506	                    detail = "RS %s listet %s nicht unter 'Satisfied by'." % (rs_id, rid)
  2507	                flagged.append({
  2508	                    "record": rid, "path": str(path), "rs_id": rs_id,
  2509	                    "issue": "upstream_not_traced",
  2510	                    "detail": detail,
  2511	                    "traced_satisfied_by": satisfied_by,
  2512	                    "provenance": {
  2513	                        "db_upstream": upstream_provenance(rec, path, rs_id),
  2514	                        "traceability": {
  2515	                            **trace_side_provenance(trace_row, rs_id, rid),
  2516	                            "matched": False,
  2517	                            "reason": reason,
  2518	                        },
  2519	                    },
  2520	                })
  2521	    for rs_id, trace_row in trace_index.items():
  2522	        for satisfier in trace_row["satisfied_by"]:
  2523	            prefix = _requirement_prefix(satisfier) if satisfier.startswith("SWS_") or "_" in satisfier else None
  2524	            candidate = None
  2525	            if prefix:
  2526	                candidate_path = RECORDS / prefix / (satisfier + ".json")
  2527	                if candidate_path.exists():
  2528	                    candidate = candidate_path
  2529	            if candidate is None:
  2530	                continue
  2531	            try:
  2532	                rec = json.loads(candidate.read_text(encoding="utf-8"))
  2533	            except (json.JSONDecodeError, OSError):
  2534	                continue
  2535	            upstream_ids = {str(u.get("id", "")).upper() for u in (rec.get("upstream") or [])
  2536	                           if isinstance(u, dict) and u.get("id")}
  2537	            if rs_id not in upstream_ids:
  2538	                flagged.append({
  2539	                    "record": satisfier, "path": str(candidate), "rs_id": rs_id,
  2540	                    "issue": "traced_not_upstream",
  2541	                    "detail": "Traceability-Tabelle nennt %s als Erfueller von %s, "
  2542	                              "Record weist dieses RS nicht als Upstream aus (hat: %s)."
  2543	                              % (satisfier, rs_id, sorted(upstream_ids) or "keins"),

  2547	                            "matched": True,

  2550	                        "db_upstream": upstream_provenance(rec, candidate, rs_id),

  2553	    return {
