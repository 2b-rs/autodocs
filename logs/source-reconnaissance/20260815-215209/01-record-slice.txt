  1341	def _record_slice(text: str, rid: str) -> str:
  1342	    """Textausschnitt ab der ID-Definition bis zum Beginn des naechsten Records.
  1343	
  1344	    Bevorzugt einen *echten* Definitionsanker ``[ID] ... ⌈`` und nicht bloss
  1345	    eine Inline-Zitierung derselben ID (z. B. in ``Dependencies``). Sonst kann
  1346	    der Slice mitten im Vorgänger-Record starten und dessen Resttext der
  1347	    gesuchten ID zuordnen.
  1348	
  1349	    Grenze ist entweder das Spec-Item-Ende (⌋) oder die naechste in eckigen
  1350	    Klammern stehende ID — sonst laufen die Eigenschaften des Folgerecords in
  1351	    den aktuellen hinein.
  1352	    """
  1353	    pattern = re.compile(r"\[%s\]" % re.escape(rid), re.IGNORECASE)
  1354	    matches = list(pattern.finditer(text))
  1355	    m = None
  1356	    for cand in matches:
  1357	        lookahead = text[cand.end():cand.end() + 240]
  1358	        if "⌈" in lookahead:
  1359	            pos = lookahead.find("⌈")
  1360	            nxt = DEF_RE.search(lookahead)
  1361	            if nxt is None or nxt.start() > pos:
  1362	                m = cand
  1363	                break
  1364	    if m is None and matches:
  1365	        m = matches[0]
  1366	    if not m:
  1367	        m = re.search(re.escape(rid), text, re.IGNORECASE)
  1368	        if not m:
  1369	            return ""
  1370	    rest = text[m.end():]
  1371	    grenzen = []
  1372	    ende = rest.find("⌋")
  1373	    if ende >= 0:
  1374	        grenzen.append(ende)
  1375	    nxt = DEF_RE.search(rest)
  1376	    if nxt:
  1377	        grenzen.append(nxt.start())
  1378	    return rest[:min(grenzen)] if grenzen else rest[:6000]
