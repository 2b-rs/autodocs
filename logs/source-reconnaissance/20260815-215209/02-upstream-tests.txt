    93	class UpstreamCliIntegrationTests(unittest.TestCase):
    94	    def fixtures(self, root):
    95	        records=root/"SWS_X"; records.mkdir(parents=True)
    96	        value={"id":"SWS_X_00001","blocks":[{"text":"Uses RS_CM_00003"}],"review":{"status":"manual"},"text_hash":"abc","upstream":[]}
    97	        path=records/"SWS_X_00001.json"; path.write_text(json.dumps(value,indent=1)+"\n",encoding="utf-8")
    98	        source={"id":"RS_CM_00003","document":"AUTOSAR_AP_RS_CommunicationManagement.pdf","page":12}
    99	        return path,value,source
   100	    def test_compare_does_not_write(self):
   101	        with tempfile.TemporaryDirectory(dir="/tmp") as td:
   102	            path,value,source=self.fixtures(Path(td)); before=path.read_bytes()
   103	            with mock.patch.object(scrape,"RECORDS",Path(td)):
   104	                report=scrape.phase_upstream({source["id"]:source},rebuild=False)
   105	            self.assertEqual(before,path.read_bytes()); self.assertEqual(report["updated"],1); self.assertEqual(report["mode"],"compare")
   106	    def test_rebuild_changes_only_upstream_and_is_idempotent(self):
   107	        with tempfile.TemporaryDirectory(dir="/tmp") as td:
   108	            path,value,source=self.fixtures(Path(td))
   109	            with mock.patch.object(scrape,"RECORDS",Path(td)):
   110	                first=scrape.phase_upstream({source["id"]:source},rebuild=True); once=path.read_bytes()
   111	                second=scrape.phase_upstream({source["id"]:source},rebuild=True)
   112	            after=json.loads(once); expected=copy.deepcopy(value); expected["upstream"]=[{"id":"RS_CM_00003","document":source["document"],"page":12,"source":"inline"}]
   113	            self.assertEqual(after,expected); self.assertEqual(first["updated"],1); self.assertEqual(second["unchanged"],1); self.assertEqual(once,path.read_bytes())
   114	    def test_missing_is_visible(self):
   115	        with tempfile.TemporaryDirectory(dir="/tmp") as td:
   116	            path,value,source=self.fixtures(Path(td))
   117	            with mock.patch.object(scrape,"RECORDS",Path(td)):
   118	                report=scrape.phase_upstream({},rebuild=False)
   119	            self.assertEqual(report["missing"],1); self.assertEqual(path.read_text(encoding="utf-8"),json.dumps(value,indent=1)+"\n")
